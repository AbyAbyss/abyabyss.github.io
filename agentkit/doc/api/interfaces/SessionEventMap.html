    Interface SessionEventMap

    Durable session facts. Augment it to add your own — a plugin that needs state to survive a reload adds an entry here and a projector, rather than hiding the state in a closure the log cannot reconstruct.

    declare module "@agentkit/runtime" {
      interface SessionEventMap {
        "review/verdict": { file: string; verdict: "pass" | "fail" };
      }
    }
    interface SessionEventMap {
        "assistant/chunk": {
            chunk: { text: string; type: "text" | "reasoning" };
        };
        "assistant/message": { content: string; toolCalls?: ToolCallRequest[] };
        "context/compacted": { summary: string; throughSeq: number };
        "context/injected": { content: string; source: string };
        error: { message: string; recoverable: boolean };
        "step/end": { step: number; usage: Usage };
        "step/start": { step: number };
        "tool/call": {
            arguments: Record<string, unknown>;
            callId: string;
            name: string;
        };
        "tool/result": {
            callId: string;
            content: string;
            isError: boolean;
            meta?: Record<string, unknown>;
            name: string;
        };
        "turn/end": { reason: TurnEndReason; turn: number };
        "turn/start": { turn: number };
        "user/message": { content: string; source?: string };
    }
